import { Injectable } from '@nestjs/common'
import Stripe from 'stripe'
import { FindOptionsOrder, FindOptionsWhere } from 'typeorm'

import { StripeProductsDetailReponse } from '@/application/admin/stripe-products/dto/stripe-products.dto'
import {
  StripeProductsPageDto,
  StripeProductsPageOptionDto,
} from '@/application/admin/stripe-products/dto/stripe-products-pagination.dto'
import { StripeProductsEntity } from '@/models/stripe-products.entity'
import { StripeProductsRepository } from '@/models/stripe-products.repository'

@Injectable()
export class StripeProductsService {
  constructor(private readonly stripeProductsRepository: StripeProductsRepository) {}
  async findAll(pageOptionsDto: StripeProductsPageOptionDto): Promise<StripeProductsPageDto> {
    const whereCondition: FindOptionsWhere<StripeProductsEntity> = {}
    const orderOption: FindOptionsOrder<StripeProductsEntity> = {}

    if (pageOptionsDto.orderBy) {
      orderOption[pageOptionsDto.orderBy] = pageOptionsDto.order
    }

    return this.stripeProductsRepository.paginationWithTransform(
      pageOptionsDto,
      StripeProductsDetailReponse,
      whereCondition,
      orderOption
    )
  }

  async createOrUpdateStripeProducts(session: Stripe.Product): Promise<void> {
    const { id: stripeProductId, name, description, metadata, active } = session
    const stripeProduct = await this.stripeProductsRepository.findOneBy({
      stripeProductId,
    })
    if (stripeProduct) {
      await this.stripeProductsRepository.update(
        {
          stripeProductId,
        },
        {
          name,
          description,
          metadata,
          active,
        }
      )
    } else {
      const stripeProduct = this.stripeProductsRepository.create({
        stripeProductId,
        name,
        description,
        metadata,
        active,
      })
      await this.stripeProductsRepository.save(stripeProduct)
    }
  }

  async deleteProduct(session: Stripe.Product): Promise<void> {
    const { id: stripeProductId } = session
    await this.stripeProductsRepository.softDelete({ stripeProductId })
  }
}
